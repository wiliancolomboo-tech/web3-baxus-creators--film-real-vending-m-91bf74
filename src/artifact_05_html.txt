  <script>
    const ws = new WebSocket("ws://localhost:8088");
    const statusLabel = document.getElementById("status-label");
    const prizeBox = document.getElementById("prize-box");
    const prizeName = document.getElementById("prize-name");
    const userDisplay = document.getElementById("user-display");

    ws.onmessage = (event) => {
      const payload = JSON.parse(event.data);
      if (payload.event === "SPIN_TRIGGERED") {
        statusLabel.innerText = "ROLLING...";
        prizeName.innerText = "DETERMINING PRIZE...";
        prizeBox.classList.remove("grail-win");
      } else if (payload.event === "DISPENSE_SUCCESS") {
        statusLabel.innerText = "DISPENSED!";
        prizeName.innerText = "GRAIL: PAPPY VAN WINKLE 15Y";
        prizeBox.classList.add("grail-win");
        userDisplay.innerText = "Telemetry: " + payload.telemetryHash.slice(0, 16) + "...";
      }
    };
