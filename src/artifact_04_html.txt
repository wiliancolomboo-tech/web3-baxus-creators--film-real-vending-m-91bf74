  <script>
    const ws = new WebSocket("ws://localhost:8088");
    const statusTitle = document.getElementById("statusTitle");
    const userAddress = document.getElementById("userAddress");
    const tierAwarded = document.getElementById("tierAwarded");
    const slotNumber = document.getElementById("slotNumber");
    const txSig = document.getElementById("txSig");

    const TIER_NAMES = ["Bourbon Rare", "Single Malt Reserve", "Cask Strength", "Pappy Van Winkle", "Macallan 25"];

    ws.onmessage = (event) => {
      const msg = JSON.parse(event.data);
      if (msg.type === "SPIN_STARTED") {
        statusTitle.innerText = "SPINNING & DISPENSING";
        statusTitle.style.color = "#D4AF37";
        userAddress.innerText = msg.user.slice(0, 6) + "..." + msg.user.slice(-4);
        tierAwarded.innerText = `${TIER_NAMES[msg.tier] || "Tier " + msg.tier}`;
        slotNumber.innerText = msg.slot;
      } else if (msg.type === "DISPENSE_CONFIRMED") {
        statusTitle.innerText = "BOTTLE DISPENSED & TRANSFERRED";
        statusTitle.style.color = "#00FFA3";
        txSig.innerText = msg.txSignature.slice(0, 10) + "..." + msg.txSignature.slice(-6);
      } else if (msg.type === "HARDWARE_JAM_ALERT") {
        statusTitle.innerText = "JAM DETECTED - ESCROW REFUND READY";
        statusTitle.style.color = "#FF4444";
      }
    };
  </script>
